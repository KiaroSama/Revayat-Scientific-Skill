% Font resolution without hand-editing: first face that exists wins.
%
% Every chain must cover Windows as well as Linux, and the LAST entry is
% unguarded — if it is missing too, fontspec aborts the build. So the
% guarded middle of each chain carries the Windows faces (Times New Roman,
% Courier New, Tahoma all ship with the OS) and the unguarded tail is the
% Linux-typical DejaVu, which those Windows guards never fall through to.
% Do not leave a Linux-only face as the last entry.
%
% Persian text face.
%
% A local fonts/ directory beside this file wins over every system name,
% and that is not a preference - it is the only way a *variable* font can
% be used at all. Asked for a family name, XeTeX hands the driver a named
% instance of the variable font, and xdvipdfmx cannot embed one: it warns
% "Invalid TTC index (not TTC font)" and dies with
% "dvipdfmx:fatal: Invalid font: -1 (4)". The identical file loaded by
% *path* carries no instance index and embeds fine. Vazirmatn now installs
% as Vazirmatn-VariableFont_wght.ttf from Google Fonts, so this is the
% common case on Windows, not an edge case.
%
% Put the files there with scripts/fetch-vazirmatn.sh (or .ps1) run in the
% document's own directory - the same fonts/ the HTML template embeds.
\IfFontExistsTF{fonts/Vazirmatn-Regular.ttf}{%
  \settextfont[Path=fonts/,BoldFont=Vazirmatn-Bold.ttf,
               Ligatures=TeX]{Vazirmatn-Regular.ttf}}{%
\IfFontExistsTF{Vazirmatn}{\settextfont[Ligatures=TeX]{Vazirmatn}}{%
  \IfFontExistsTF{Shabnam}{\settextfont[Ligatures=TeX]{Shabnam}}{%
    \IfFontExistsTF{Sahel}{\settextfont[Ligatures=TeX]{Sahel}}{%
      \IfFontExistsTF{Amiri}{\settextfont[Ligatures=TeX]{Amiri}}{%
        \IfFontExistsTF{Noto Naskh Arabic}{%
          \settextfont[Ligatures=TeX]{Noto Naskh Arabic}}{%
          \IfFontExistsTF{Tahoma}{\settextfont[Ligatures=TeX]{Tahoma}}{%
            \settextfont[Ligatures=TeX]{DejaVu Sans}}}}}}}}
% Latin text face for isolated English and Western digits.
%
% OS-native faces come FIRST, and that ordering is not cosmetic. On MiKTeX
% \IfFontExistsTF answers "yes" for a face that is not installed: MiKTeX
% tries to manufacture a METAFONT font for it, the name gets truncated
% ("TeX Gyre Term.cfg", "Liberation Ser.cfg"), makemf fails, and the run
% dies later in the driver with "dvipdfmx:fatal: Invalid font: -1 (4)".
% Testing a real system font first never reaches that path. On Linux,
% fontconfig answers honestly, so the TeX Gyre / DejaVu tail still wins.
\IfFontExistsTF{Times New Roman}{%
  \setlatintextfont{Times New Roman}}{%
  \IfFontExistsTF{TeX Gyre Termes}{%
    \setlatintextfont{TeX Gyre Termes}}{%
    \IfFontExistsTF{Liberation Serif}{%
      \setlatintextfont{Liberation Serif}}{%
      \setlatintextfont{DejaVu Serif}}}}
% Monospace for listings. Same ordering rule.
\IfFontExistsTF{Consolas}{\setmonofont[Scale=0.92]{Consolas}}{%
  \IfFontExistsTF{Courier New}{\setmonofont[Scale=0.92]{Courier New}}{%
    \IfFontExistsTF{TeX Gyre Cursor}{\setmonofont[Scale=0.92]{TeX Gyre Cursor}}{%
      \IfFontExistsTF{Liberation Mono}{\setmonofont[Scale=0.90]{Liberation Mono}}{%
        \setmonofont[Scale=0.90]{DejaVu Sans Mono}}}}}

% xepersian requires U+06F0 in the digit font. Keep numeric runs in \en.
% Resolve the local file first to avoid variable-font named instances on MiKTeX.
\IfFontExistsTF{fonts/Vazirmatn-Regular.ttf}{%
  \setdigitfont[Path=fonts/]{Vazirmatn-Regular.ttf}}{%
  \IfFontExistsTF{Vazirmatn}{\setdigitfont{Vazirmatn}}{%
    \IfFontExistsTF{Tahoma}{\setdigitfont{Tahoma}}{%
      \IfFontExistsTF{Amiri}{\setdigitfont{Amiri}}{%
        \setdigitfont{DejaVu Sans}}}}}
